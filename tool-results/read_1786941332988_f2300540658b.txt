   106→function generateCodigoUnico() {
   107→  // Codigo único basado en timestamp + random
   108→  return `EST-${Date.now().toString(36).toUpperCase()}${Math.random()
   109→    .toString(36)
   110→    .slice(2, 6)
   111→    .toUpperCase()}`
   112→}

   114→interface StudentFormValues {
   115→  nombre: string
   116→  apellido: string
   117→  cedulaEscolar: string
   118→  sectionId: string
   119→  genero: string
   120→  fechaNacimiento: string
   121→}
   122→
   123→function emptyForm(sectionDefault = ''): StudentFormValues {
   124→  return {
   125→    nombre: '',
   126→    apellido: '',
   127→    cedulaEscolar: '',
   128→    sectionId: sectionDefault,
   129→    genero: '',
   130→    fechaNacimiento: '',
   131→  }
   132→}
   133→
   134→export function StudentsManager() {
   135→  const [students, setStudents] = useState<StudentRow[]>([])
   136→  const [sections, setSections] = useState<SectionOption[]>([])
   137→  const [loading, setLoading] = useState(true)
   138→  const [search, setSearch] = useState('')
   139→  const [sectionFilter, setSectionFilter] = useState<string>('all')
   140→  const [dialogOpen, setDialogOpen] = useState(false)
   141→  const [editing, setEditing] = useState<StudentRow | null>(null)
   142→  const [form, setForm] = useState<StudentFormValues>(emptyForm())
   143→  const [submitting, setSubmitting] = useState(false)
   144→  const [togglingId, setTogglingId] = useState<string | null>(null)
   145→
   146→  // Photo upload (in edit dialog)
   147→  const [photoFile, setPhotoFile] = useState<File | null>(null)
   148→  const [photoPreview, setPhotoPreview] = useState<string | null>(null)
   149→  const [uploadingPhoto, setUploadingPhoto] = useState(false)
   150→  const photoInputRef = useRef<HTMLInputElement>(null)
   151→
   152→  // Edit double confirmation
   153→  const [confirmEditOpen, setConfirmEditOpen] = useState(false)
   154→
   155→  // Delete double confirmation
   156→  const [deleteTarget, setDeleteTarget] = useState<StudentRow | null>(null)
   157→  const [deleteConfirmName, setDeleteConfirmName] = useState('')
   158→  const [deleteStep2, setDeleteStep2] = useState(false)
   159→  const [deleting, setDeleting] = useState(false)
   160→
   161→  // Carnet PDF loading
   162→  const [pdfLoadingId, setPdfLoadingId] = useState<string | null>(null)
   163→
   164→  // Monthly attendance PDF dialog
   165→  const [attendancePdfTarget, setAttendancePdfTarget] = useState<StudentRow | null>(null)
   166→  const todayForMonth = new Date()
   167→  const defaultMonth = `${todayForMonth.getFullYear()}-${String(
   168→    todayForMonth.getMonth() + 1
   169→  ).padStart(2, '0')}`
   170→  const [attendanceMonth, setAttendanceMonth] = useState(defaultMonth)
   171→
   172→  const openAttendancePdf = async (s: StudentRow) => {
   173→    if (!attendanceMonth || !/^\d{4}-\d{2}$/.test(attendanceMonth)) {
   174→      toast.error('Selecciona un mes válido (YYYY-MM)')
   175→      return
   176→    }
   177→    setPdfLoadingId(s.id)
   178→    try {
   179→      const { useAuthStore } = await import('@/stores/auth-store')
   180→      const token = useAuthStore.getState().token
   181→      const url = `/api/admin/students/${s.id}/attendance-pdf?month=${attendanceMonth}${
   182→        token ? `&token=${encodeURIComponent(token)}` : ''
   183→      }`
   184→      window.open(url, '_blank', 'noopener,noreferrer')
   185→      setAttendancePdfTarget(null)
   186→      toast.success('Generando reporte de asistencia…')
   187→    } catch (e: any) {
   188→      toast.error('No se pudo generar el reporte: ' + (e.message || ''))
   189→    } finally {
   190→      setTimeout(() => setPdfLoadingId(null), 800)
   191→    }
   192→  }
   193→
   194→  const loadSections = useCallback(async () => {
   195→    try {
   196→      const res = await api.get<{ data: SectionOption[] }>('/admin/sections')
   197→      setSections(res.data || [])
   198→    } catch (e: any) {
   199→      toast.error('Error al cargar secciones', { description: e.message })
   200→    }
   201→  }, [])
   202→
   203→  const loadStudents = useCallback(async () => {
   204→    setLoading(true)
   205→    try {
   206→      const params = new URLSearchParams()
   207→      params.set('limit', '100')
   208→      if (search) params.set('search', search)
   209→      if (sectionFilter !== 'all') params.set('sectionId', sectionFilter)
   210→      params.set('includeInactive', 'true')
   211→      const res = await api.get<StudentsResponse>(`/admin/students?${params.toString()}`)
   212→      setStudents(res.data || [])
   213→    } catch (e: any) {
   214→      toast.error('Error al cargar estudiantes', { description: e.message })
   215→    } finally {
   216→      setLoading(false)
   217→    }
   218→  }, [search, sectionFilter])
   219→
   220→  useEffect(() => {
   221→    loadSections()
   222→  }, [loadSections])
   223→
   224→  useEffect(() => {
   225→    const t = setTimeout(loadStudents, 250)
   226→    return () => clearTimeout(t)
   227→  }, [loadStudents])
   228→
   229→  const resetPhotoState = () => {
   230→    setPhotoFile(null)
   231→    setPhotoPreview(null)
   232→    setUploadingPhoto(false)
   233→    if (photoInputRef.current) photoInputRef.current.value = ''
   234→  }
   235→
   236→  const openCreate = () => {
   237→    setEditing(null)
   238→    setForm(emptyForm(sections[0]?.id || ''))
   239→    resetPhotoState()
   240→    setDialogOpen(true)
   241→  }
   242→
   243→  const openEdit = (s: StudentRow) => {
   244→    setEditing(s)
   245→    setForm({
   246→      nombre: s.nombre,
   247→      apellido: s.apellido,
   248→      cedulaEscolar: s.cedulaEscolar || '',
   249→      sectionId: s.sectionId,
   250→      genero: s.genero || '',
   251→      fechaNacimiento: s.fechaNacimiento || '',
   252→    })
   253→    resetPhotoState()
   254→    setPhotoPreview(s.fotoKey ? `/api/files/${s.fotoKey}` : null)
   255→    setDialogOpen(true)
   256→  }
   257→
   258→  const handlePhotoSelect = (e: React.ChangeEvent<HTMLInputElement>) => {
   259→    const file = e.target.files?.[0]
   260→    if (!file) return
   261→    if (!file.type.startsWith('image/')) {
   262→      toast.error('Solo se permiten imágenes')
   263→      return
   264→    }
   265→    if (file.size > 5 * 1024 * 1024) {
   266→      toast.error('La imagen no debe superar 5MB')
   267→      return
   268→    }
   269→    setPhotoFile(file)
   270→    const url = URL.createObjectURL(file)
   271→    setPhotoPreview(url)
   272→  }
   273→
   274→  // Step 1 of edit: validate form → open second confirmation dialog
   275→  const handleSubmitClick = (e: React.FormEvent) => {
   276→    e.preventDefault()
   277→    if (!form.nombre || !form.apellido || !form.sectionId) {
   278→      toast.error('Nombre, apellido y sección son obligatorios')
   279→      return
   280→    }
   281→    setConfirmEditOpen(true)
   282→  }
   283→
   284→  // Step 2 of edit: actually save (after second confirmation)
   285→  const performSave = async () => {
   286→    setSubmitting(true)
   287→    try {
   288→      // 1. If a new photo was selected, upload it first
   289→      let newFotoKey: string | null = null
   290→      if (photoFile && editing) {
   291→        setUploadingPhoto(true)
   292→        try {
   293→          const token = useAuthStore.getState().token
   294→          const formData = new FormData()
   295→          formData.append('file', photoFile)
   296→          formData.append('estudianteId', editing.id)
   297→          // Use plain fetch (NOT apiFetch) so the browser sets the correct
   298→          // multipart/form-data Content-Type automatically with the boundary.
   299→          const uploadRes = await fetch('/api/alumno/photo', {
   300→            method: 'POST',
   301→            headers: token ? { Authorization: `Bearer ${token}` } : {},
   302→            body: formData,
   303→          })
   304→          if (!uploadRes.ok) {
   305→            const errBody = await uploadRes.json().catch(() => ({}))
   306→            throw new Error((errBody as any).error || `HTTP ${uploadRes.status}`)
   307→          }
   308→          const data = (await uploadRes.json()) as { mediaKey: string }
   309→          newFotoKey = data.mediaKey
   310→        } catch (err: any) {
   311→          toast.error('No se pudo subir la foto: ' + err.message)
   312→          setSubmitting(false)
   313→          setUploadingPhoto(false)
   314→          return
   315→        } finally {
   316→          setUploadingPhoto(false)
   317→        }
   318→      }
   319→
   320→      // 2. PUT the student data
   321→      if (editing) {
   322→        await api.put(`/admin/students/${editing.id}`, {
   323→          nombre: form.nombre,
   324→          apellido: form.apellido,
   325→          cedulaEscolar: form.cedulaEscolar || null,
   326→          sectionId: form.sectionId,
   327→          genero: form.genero || null,
   328→          fechaNacimiento: form.fechaNacimiento || null,
   329→        })
   330→        toast.success('Estudiante actualizado')
   331→      } else {
   332→        await api.post('/admin/students', {
   333→          codigoUnico: generateCodigoUnico(),
   334→          nombre: form.nombre,
   335→          apellido: form.apellido,
   336→          cedulaEscolar: form.cedulaEscolar || null,
   337→          sectionId: form.sectionId,
   338→          genero: form.genero || null,
   339→          fechaNacimiento: form.fechaNacimiento || null,
   340→        })
   341→        toast.success('Estudiante creado')
   342→      }
   343→
   344→      // If photo uploaded for a brand-new student, we'd need the created id — for now only edit supports photo upload (per spec)
   345→      if (newFotoKey) {
   346→        toast.success('Foto actualizada')
   347→      }
   348→
   349→      setConfirmEditOpen(false)
   350→      setDialogOpen(false)
   351→      resetPhotoState()
   352→      loadStudents()
   353→    } catch (e: any) {
   354→      toast.error(e.message || 'Error al guardar estudiante')
   355→    } finally {
   356→      setSubmitting(false)
   357→    }
   358→  }
   359→
   360→  // === Delete flow (double confirmation) ===
   361→  const openDeleteStep1 = (s: StudentRow) => {
   362→    setDeleteTarget(s)
   363→    setDeleteConfirmName('')
   364→    setDeleteStep2(false)
   365→  }
   366→
   367→  const advanceToDeleteStep2 = () => {
   368→    setDeleteStep2(true)
   369→    setDeleteConfirmName('')
   370→  }
   371→
   372→  const cancelDelete = () => {
   373→    setDeleteTarget(null)
   374→    setDeleteStep2(false)
   375→    setDeleteConfirmName('')
   376→  }
   377→
   378→  const performDelete = async () => {
   379→    if (!deleteTarget) return
   380→    const expected = `${deleteTarget.nombre} ${deleteTarget.apellido}`.trim().toLowerCase()
   381→    if (deleteConfirmName.trim().toLowerCase() !== expected) {
   382→      toast.error('El nombre no coincide')
   383→      return
   384→    }
   385→    setDeleting(true)
   386→    try {
   387→      await api.delete(`/admin/students/${deleteTarget.id}`)
   388→      toast.success('Estudiante eliminado')
   389→      cancelDelete()
   390→      loadStudents()
   391→    } catch (e: any) {
   392→      toast.error(e.message || 'Error al eliminar')
   393→    } finally {
   394→      setDeleting(false)
   395→    }
   396→  }
   397→
   398→  const handleToggleActivo = async (s: StudentRow) => {
   399→    setTogglingId(s.id)
   400→    try {
   401→      await api.put(`/admin/students/${s.id}`, { activo: !s.activo })
   402→      setStudents((prev) =>
   403→        prev.map((row) => (row.id === s.id ? { ...row, activo: !row.activo } : row))
   404→      )
   405→      toast.success(s.activo ? 'Estudiante desactivado' : 'Estudiante activado')
   406→    } catch (e: any) {
   407→      toast.error(e.message || 'Error al cambiar estado')
   408→    } finally {
   409→      setTogglingId(null)
   410→    }
   411→  }
   412→
   413→  // === Carnet PDF download ===
   414→  const openCarnetPdf = (s: StudentRow) => {
   415→    setPdfLoadingId(s.id)
   416→    try {
   417→      const token = useAuthStore.getState().token
   418→      const url = `/api/admin/students/${s.id}/carnet-pdf${token ? `?token=${encodeURIComponent(token)}` : ''}`
   419→      window.open(url, '_blank', 'noopener,noreferrer')
   420→    } catch (e: any) {
   421→      toast.error('No se pudo generar el carnet PDF: ' + (e.message || ''))
   422→    } finally {
   423→      setTimeout(() => setPdfLoadingId(null), 800)
   424→    }
   425→  }
   426→
   427→  const sectionName = useMemo(() => {
   428→    const map = new Map(sections.map((s) => [s.id, s]))
   429→    return (id: string) => {
   430→      const s = map.get(id)
   431→      return s ? `${s.nombre} · ${turnoLabel[s.turno] || s.turno}` : '—'
   432→    }
   433→  }, [sections])
   434→
   435→  const initialsOf = (s: StudentRow) =>
   436→    `${s.nombre?.[0] || ''}${s.apellido?.[0] || ''}`.toUpperCase()
   437→
   438→  return (
   439→    <div className="space-y-6">
   440→      <div className="flex items-start justify-between gap-4 flex-wrap">
   441→        <div>
   442→          <h2 className="text-2xl font-bold tracking-tight">Estudiantes</h2>
   443→          <p className="text-sm text-muted-foreground">
   444→            Gestiona el registro de estudiantes del plantel
   445→          </p>
   446→        </div>
   447→        <Button onClick={openCreate} className="bg-emerald-600 hover:bg-emerald-700 text-white">
   448→          <Plus className="w-4 h-4" />
   449→          Agregar estudiante
   450→        </Button>
   451→      </div>
   452→
   453→      <Card>
   454→        <CardHeader>
   455→          <div className="flex items-center gap-3 flex-wrap">
   456→            <div className="relative flex-1 min-w-[200px]">
   457→              <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   458→              <Input
   459→                placeholder="Buscar por nombre o código..."
   460→                value={search}
   461→                onChange={(e) => setSearch(e.target.value)}
   462→                className="pl-9"
   463→              />
   464→            </div>
   465→            <Select value={sectionFilter} onValueChange={setSectionFilter}>
   466→              <SelectTrigger className="w-[200px]">
   467→                <SelectValue placeholder="Todas las secciones" />
   468→              </SelectTrigger>
   469→              <SelectContent>
   470→                <SelectItem value="all">Todas las secciones</SelectItem>
   471→                {sections.map((s) => (
   472→                  <SelectItem key={s.id} value={s.id}>
   473→                    {s.nombre} · {turnoLabel[s.turno] || s.turno}
   474→                  </SelectItem>
   475→                ))}
   476→              </SelectContent>
   477→            </Select>
   478→          </div>
   479→        </CardHeader>
   480→        <CardContent>
   481→          {loading ? (
   482→            <div className="space-y-2">
   483→              {Array.from({ length: 5 }).map((_, i) => (
   484→                <Skeleton key={i} className="h-10 w-full" />
   485→              ))}
   486→            </div>
   487→          ) : students.length === 0 ? (
   488→            <div className="py-12 text-center">
   489→              <GraduationCap className="w-10 h-10 mx-auto text-muted-foreground/50" />
   490→              <p className="mt-3 text-sm text-muted-foreground">
   491→                No hay estudiantes registrados
   492→              </p>
   493→              <p className="text-xs text-muted-foreground">
   494→                Comienza agregando un estudiante o ajusta los filtros
   495→              </p>
   496→            </div>
   497→          ) : (
   498→            <div className="max-h-[60vh] overflow-y-auto -mx-2">
   499→              <Table>
   500→                <TableHeader>
   501→                  <TableRow>
   502→                    <TableHead>Estudiante</TableHead>
   503→                    <TableHead className="hidden md:table-cell">Cédula escolar</TableHead>
   504→                    <TableHead>Sección</TableHead>
   505→                    <TableHead className="hidden sm:table-cell">Género</TableHead>
   506→                    <TableHead className="text-center">Activo</TableHead>
   507→                    <TableHead className="text-right">Acciones</TableHead>
   508→                  </TableRow>
   509→                </TableHeader>
   510→                <TableBody>
   511→                  {students.map((s) => (
   512→                    <TableRow key={s.id} className={!s.activo ? 'opacity-60' : ''}>
   513→                      <TableCell>
   514→                        <div className="flex items-center gap-3">
   515→                          <Avatar className="w-9 h-9 border border-emerald-100 dark:border-emerald-950 shrink-0">
   516→                            {s.fotoKey ? (
   517→                              <AvatarImage
   518→                                src={`/api/files/${s.fotoKey}`}
   519→                                alt={`${s.nombre} ${s.apellido}`}
   520→                              />
   521→                            ) : null}
   522→                            <AvatarFallback className="bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400 text-xs font-semibold">
   523→                              {initialsOf(s)}
   524→                            </AvatarFallback>
   525→                          </Avatar>
   526→                          <div className="min-w-0">
   527→                            <p className="font-medium truncate">
   528→                              {s.apellido}, {s.nombre}
   529→                            </p>
   530→                            <p className="text-xs text-muted-foreground font-mono">
   531→                              {s.codigoUnico}
   532→                            </p>
   533→                          </div>
   534→                        </div>
   535→                      </TableCell>
   536→                      <TableCell className="hidden md:table-cell">
   537→                        <span className="text-sm">{s.cedulaEscolar || '—'}</span>
   538→                      </TableCell>
   539→                      <TableCell>
   540→                        {s.section ? (
   541→                          <Badge variant="outline" className="font-normal">
   542→                            {sectionName(s.section.id)}
   543→                          </Badge>
   544→                        ) : (
   545→                          '—'
   546→                        )}
   547→                      </TableCell>
   548→                      <TableCell className="hidden sm:table-cell">
   549→                        {s.genero ? generoLabel[s.genero] || s.genero : '—'}
   550→                      </TableCell>
   551→                      <TableCell className="text-center">
   552→                        <Switch
   553→                          checked={s.activo}
   554→                          onCheckedChange={() => handleToggleActivo(s)}
   555→                          disabled={togglingId === s.id}
   556→                        />
   557→                      </TableCell>
   558→                      <TableCell className="text-right">
   559→                        <div className="flex justify-end gap-1">
   560→                          <Button
   561→                            size="icon"
   562→                            variant="ghost"
   563→                            onClick={() => openEdit(s)}
   564→                            aria-label="Editar"
   565→                            title="Editar"
   566→                          >
   567→                            <Pencil className="w-4 h-4" />
   568→                          </Button>
   569→                          <Button
   570→                            size="icon"
   571→                            variant="ghost"
   572→                            onClick={() => openCarnetPdf(s)}
   573→                            disabled={pdfLoadingId === s.id}
